# Question  
```
Suppose LeetCode will start its IPO soon. In order to sell a good price of its shares to Venture Capital, LeetCode would like to work on some 
projects to increase its capital before the IPO. Since it has limited resources, it can only finish at most k distinct projects before the IPO. 
Help LeetCode design the best way to maximize  
its total capital after finishing at most k distinct projects.
You are given n projects where the ith project has a pure profit profits[i] and a minimum capital of capital[i] is needed to start it.

Initially, you have w capital. When you finish a project, you will obtain its pure profit and the profit will be added to your total capital.

Pick a list of at most k distinct projects from given projects to maximize your final capital, and return the final maximized capital.

The answer is guaranteed to fit in a 32-bit signed integer.
```
# Constraints  
```
1 <= k <= 1e5
0 <= w <= 1e9
n == profits.length
n == capital.length
1 <= n <= 1e5
0 <= profits[i] <= 1e4
0 <= capital[i] <= 1e9
```
# Approach Used  
1. Sorting based on capital.
2. Segment-tree + Range Query
4. Range updation based on capital.

# Code (in C++)  
```
class Solution {
public:

    void build(int node, int start, int end,
               vector<pair<int,int>>& v,
               vector<pair<int,int>>& segtree) {

        if(start == end) {
            segtree[node] = {v[start].second, start};
            return;
        }

        int mid = start + (end - start) / 2;

        build(2 * node + 1, start, mid, v, segtree);
        build(2 * node + 2, mid + 1, end, v, segtree);

        segtree[node] = max(segtree[2 * node + 1],
                            segtree[2 * node + 2]);
    }

    pair<int,int> find_max(int node, int start, int end,
                           int l, int r,
                           vector<pair<int,int>>& segtree,
                           vector<int>& profits) {
        if(r < start || end < l)
            return {INT_MIN, -1};

        if(l <= start && end <= r)
            return segtree[node];

        int mid = start + (end - start) / 2;

        pair<int,int> left =
            find_max(2 * node + 1, start, mid,
                     l, r, segtree, profits);

        pair<int,int> right =
            find_max(2 * node + 2, mid + 1, end,
                     l, r, segtree, profits);

        return max(left, right);
    }

    void rebuild(int node, int start, int end,
                 int idx,
                 vector<pair<int,int>>& segtree,
                 vector<int>& profits) {

        if(start == end) {
            segtree[node] = {INT_MIN, -1};
            return;
        }

        int mid = start + (end - start) / 2;

        if(idx <= mid)
            rebuild(2 * node + 1, start, mid,
                    idx, segtree, profits);
        else
            rebuild(2 * node + 2, mid + 1, end,
                    idx, segtree, profits);

        segtree[node] = max(segtree[2 * node + 1],
                            segtree[2 * node + 2]);
    }

    int findMaximizedCapital(int k, int w,
                             vector<int>& profits,
                             vector<int>& capital) {

        int n = profits.size();

        vector<pair<int,int>> v;

        for(int i = 0; i < n; ++i)
            v.push_back({capital[i], profits[i]});
        sort(v.begin(), v.end());

        vector<pair<int,int>> segtree(4 * n,
                                      {INT_MIN, -1});

        build(0, 0, n - 1, v, segtree);

        for(int i = 0; i < k; ++i) {

            int start = 0;
            int end = n - 1;
            int ans = -1;

            while(start <= end) {

                int mid = start + (end - start) / 2;

                if(v[mid].first <= w) {
                    ans = mid;
                    start = mid + 1;
                }
                else {
                    end = mid - 1;
                }
            }
            if(ans == -1)
                return w;

    
            pair<int,int> p =
                find_max(0, 0, n - 1,
                         0, ans,
                         segtree, profits);

            int val = p.first;
            int idx = p.second;
            if(val<0) return w;
            w += val;
            rebuild(0, 0, n - 1,
                    idx, segtree, profits);
        }

        return w;
    }
};
```
